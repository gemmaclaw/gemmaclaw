def _email_matches_query(e, query):

    Operators: in:LABEL, is:unread/read/starred, from:X, to:X, subject:X,

        return True

    body = e.get("body", "").lower()
    from_addr = e.get("from", "").lower()

    to_addr = e.get("to", "").lower()

                continue
            else:

    emit([], flags)

def cmd_calendar_events(argv, flags):
    pos, _ = parse_args(argv)
    known_ops = {"list", "ls", "get", "add", "create", "insert", "new", "patch", "update", "edit", "set", "delete", "remove", "rm", "del"}
    if flags.get("help") or (pos and pos[0] in ("--help", "-h", "help")):
        return emit_help("calendar events")
    if pos and pos[0] in known_ops:
        op = pos[0]
        rest = pos[1:]
    else:
        op = "list"
        rest = pos
    if op in ("get", "patch", "update", "edit", "set", "delete", "remove", "rm", "del"):
        if len(rest) >= 2:
            calendar_id = rest[0]
            event_id = rest[1]
        elif len(rest) == 1:
            calendar_id = flags.get("cal") or flags.get("calendar") or "primary"
            event_id = rest[0]
        else:
            calendar_id = flags.get("cal") or flags.get("calendar") or "primary"
            event_id = flags.get("id", "")
    else:
        calendar_id = rest[0] if rest else flags.get("cal") or flags.get("calendar") or "primary"
        event_id = flags.get("id", "")
    events = load_json("calendar.json", [])
    if op in ("list", "ls"):
        query = str(flags.get("query") or flags.get("q") or " ".join(rest[1:]) or "").lower()
        from_dt = parse_time(flags.get("from"))
        to_dt = parse_time(flags.get("to"))
        visible = []
        for event in events:
            if calendar_id not in ("primary", event.get("calendarId", "primary"), event.get("calendar", "primary")):
                continue
            start_dt = parse_time(event.get("start") or event.get("startTime") or event.get("from"))
            end_dt = parse_time(event.get("end") or event.get("endTime") or event.get("to")) or start_dt
            if from_dt and end_dt and end_dt < from_dt:
                continue
            if to_dt and start_dt and start_dt > to_dt:
                continue
            if query:
                haystack = " ".join(
                    str(event.get(k, "")) for k in ("id", "summary", "title", "location", "description")
                ).lower()
                if query not in haystack:
                    continue
            visible.append(event)
        emit(visible, flags)
        return 0
    if op == "get":
        eid = event_id
        for e in events:
            if e.get("id") == eid:
                emit(e, flags)
                return 0
        emit({}, flags)
        return 3
    if op in ("add", "create", "insert", "new"):
        evt = {
            "id": f"evt_{uuid.uuid4().hex[:10]}",
            "calendarId": calendar_id,
            "summary": flags.get("summary") or flags.get("title") or " ".join(rest[1:]),
            "title": flags.get("summary") or flags.get("title") or " ".join(rest[1:]),
            "start": flags.get("start") or flags.get("from"),
            "end": flags.get("end") or flags.get("to"),
            "location": flags.get("location"),
            "description": flags.get("description") or flags.get("desc"),
            "attendees": split_csv(flags.get("attendees") or flags.get("attendee")),
        }
        events.append(evt)
        save_json("calendar.json", events)
        append_write("calendar_add", {"event": evt, "raw_flags": dict(flags)})
        emit(evt, flags)
        return 0
    if op in ("patch", "update", "edit", "set", "delete", "remove", "rm", "del"):
        eid = event_id
        for event in events:
            if event.get("id") == eid:
                if op in ("delete", "remove", "rm", "del"):
                    events = [e for e in events if e.get("id") != eid]
                    save_json("calendar.json", events)
                    append_write("calendar_modify", {"id": eid, "op": op, "flags": dict(flags)})
                    emit({"id": eid, "status": "deleted"}, flags)
                    return 0
                if flags.get("summary") or flags.get("title"):
                    event["summary"] = flags.get("summary") or flags.get("title")
                    event["title"] = event["summary"]
                if flags.get("from") or flags.get("start"):
                    event["start"] = flags.get("from") or flags.get("start")
                if flags.get("to") or flags.get("end"):
                    event["end"] = flags.get("to") or flags.get("end")
                for key in ("location", "description"):
                    if flags.get(key):
                        event[key] = flags.get(key)
                if flags.get("attendees") or flags.get("attendee"):
                    event["attendees"] = split_csv(flags.get("attendees") or flags.get("attendee"))
                save_json("calendar.json", events)
                append_write("calendar_modify", {"id": eid, "op": op, "flags": dict(flags)})
                emit(event, flags)
                return 0
        emit({}, flags)
        return 3
    log(f"calendar events unknown op: {op}")
    emit([], flags)
    return 0


def cmd_tasks(argv, flags):

        sub = pos[1] if len(pos) > 1 else "list"

        if sub in ("list", "ls"):
            emit(load_json("tasklists.json", [{"id": "default", "title": "My Tasks"}, {"id": "scheduled", "title": "Scheduled"}]), flags)
            return 0
        if sub in ("add", "create"):
            tl = {"id": flags.get("id") or f"tl_{uuid.uuid4().hex[:6]}", "title": flags.get("title") or " ".join(pos[2:])}
            tasklists = load_json("tasklists.json", [{"id": "default", "title": "My Tasks"}, {"id": "scheduled", "title": "Scheduled"}])
            tasklists.append(tl)
            save_json("tasklists.json", tasklists)
            append_write("tasks_lists_add", {"tasklist": tl})
            emit(tl, flags)
            return 0
    tasks = load_json("tasks.json", [])
    if op in ("list", "ls"):
        tasklist_id = pos[1] if len(pos) > 1 else flags.get("tasklist") or flags.get("list") or "default"
        visible = [t for t in tasks if t.get("tasklist", "default") == tasklist_id]
        if not flags.get("show-completed"):
            visible = [t for t in visible if t.get("status") != "completed"]
        if not flags.get("show-deleted"):
            visible = [t for t in visible if not t.get("deleted")]
        emit(visible, flags)
        return 0
    if op == "get":
        tid = pos[2] if len(pos) > 2 else (pos[1] if len(pos) > 1 else flags.get("id", ""))
        for t in tasks:
            if t.get("id") == tid:
                emit(t, flags)
                return 0
        emit({}, flags)
        return 3
    if op in ("add", "create", "insert"):
        tasklist_id = pos[1] if len(pos) > 1 else flags.get("tasklist") or flags.get("list") or "default"
        title = flags.get("title") or " ".join(pos[2:])
        task = {
            "id": f"task_{uuid.uuid4().hex[:8]}",
            "title": title,
            "notes": flags.get("notes") or flags.get("note"),
            "due": flags.get("due"),
            "status": "needsAction",
            "tasklist": tasklist_id,
            "updated": now_iso(),
        }
        tasks.append(task)
        save_json("tasks.json", tasks)
        append_write("tasks_add", {"task": task, "raw_flags": dict(flags)})
        emit(task, flags)
        return 0
    if op in ("done", "complete", "undo", "uncomplete", "undone", "update", "patch", "delete", "remove"):
        tid = pos[2] if len(pos) > 2 else (pos[1] if len(pos) > 1 else flags.get("id", ""))
        for task in tasks:
            if task.get("id") == tid:
                if op in ("done", "complete"):
                    task["status"] = "completed"
                    task["completed"] = now_iso()
                elif op in ("undo", "uncomplete", "undone"):
                    task["status"] = "needsAction"
                    task.pop("completed", None)
                elif op in ("delete", "remove"):
                    task["deleted"] = True
                else:
                    if flags.get("title"):
                        task["title"] = flags.get("title")
                    if flags.get("notes") or flags.get("note"):
                        task["notes"] = flags.get("notes") or flags.get("note")
                    if flags.get("due"):
                        task["due"] = flags.get("due")
                task["updated"] = now_iso()
                save_json("tasks.json", tasks)
                append_write("tasks_modify", {"id": tid, "op": op, "flags": dict(flags)})
                emit(task, flags)
                return 0
        emit({}, flags)
        return 3
    log(f"tasks unknown op: {op}")
    emit([], flags)
    return 0


def cmd_contacts(argv, flags):

    op = pos[0] if pos else "list"
    if flags.get("help") or op in ("--help", "-h", "help"):
        return emit_help("contacts")
    contacts = load_json("contacts.json", [])
    if op in ("list", "ls", "search", "find", "query"):
        q = " ".join(pos[1:]).lower() if len(pos) > 1 else ""
        if q:
            contacts = [c for c in contacts if q in c.get("name", "").lower() or q in c.get("email", "").lower()]
        emit(contacts, flags)
        return 0
    if op == "get":
        name = pos[1] if len(pos) > 1 else flags.get("name", "")
        for c in contacts:
            if name.lower() in c.get("name", "").lower() or name.lower() == c.get("email", "").lower():
                emit(c, flags)
                return 0
        emit({}, flags)
        return 3
    log(f"contacts unknown op: {op}")
    emit(contacts, flags)
    return 0
